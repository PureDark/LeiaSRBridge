cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build: cmake --preset windows-x64")
endif()
project(TitanLeiaBridge VERSION 0.4.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "TitanLeiaBridge requires Windows x64 MSVC (Leia SDK ABI)")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
add_compile_definitions(UNICODE _UNICODE "WIN32_LEAN_AND_MEAN=" NOMINMAX)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/EHsc;/utf-8;/Zc:__cplusplus>")
add_compile_options("$<$<CONFIG:Release>:/Zi>")

set(VENDOR_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/vendor" CACHE PATH "Prepared portable vendor runtime")
foreach(required "${LEIA_SDK_ROOT}/include" "${OPENCV_RUNTIME}"
    "${VENDOR_RUNTIME_ROOT}/MCU_library.dll" "${VENDOR_RUNTIME_ROOT}/eyetracking/FaceTracking.exe")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Missing local dependency: ${required}. See docs/BUILD.zh-CN.md")
  endif()
endforeach()
set(PACKAGE_DIR "${CMAKE_BINARY_DIR}/package")
# Stage the prepared runtime; never overwrite our rebuilt result proxy or copy device calibration.
file(COPY "${VENDOR_RUNTIME_ROOT}/" DESTINATION "${PACKAGE_DIR}/vendor"
  PATTERN "BOE_3DET_SDK.dll" EXCLUDE PATTERN "boe_mcu_mat.ini" EXCLUDE
  PATTERN "logs" EXCLUDE PATTERN "*.log" EXCLUDE)
set(RUNTIME_DIR "${PACKAGE_DIR}/runtime")
set(TEST_DIR "${CMAKE_BINARY_DIR}/tests")
set(CMAKE_PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/symbols")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_PDB_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/symbols")
endforeach()
function(output_in target dir)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${dir}" LIBRARY_OUTPUT_DIRECTORY "${dir}")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${dir}" LIBRARY_OUTPUT_DIRECTORY_${config} "${dir}")
  endforeach()
endfunction()
