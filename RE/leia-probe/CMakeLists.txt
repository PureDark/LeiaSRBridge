cmake_minimum_required(VERSION 3.25)
project(LeiaReadOnlyProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../dependencies/LeiaSDK" CACHE PATH "Leia SDK 1.34.10 x64")
add_executable(LeiaReadOnlyProbe probe.cpp)
target_compile_definitions(LeiaReadOnlyProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LeiaReadOnlyProbe PRIVATE /EHsc /utf-8)
target_include_directories(LeiaReadOnlyProbe PRIVATE "${SDK}/include" "${SDK}/third_party/OpenCV/include")
target_link_directories(LeiaReadOnlyProbe PRIVATE "${SDK}/lib")
target_link_libraries(LeiaReadOnlyProbe PRIVATE SimulatedRealityCore SimulatedRealityFaceTrackers psapi "${SDK}/third_party/OpenCV/lib/x64/opencv_world343.lib")
add_executable(LeiaProbeMathCheck math_check.cpp)
target_compile_definitions(LeiaProbeMathCheck PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LeiaProbeMathCheck PRIVATE /EHsc /utf-8)
target_include_directories(LeiaProbeMathCheck PRIVATE "${SDK}/include" "${SDK}/third_party/OpenCV/include")
target_link_directories(LeiaProbeMathCheck PRIVATE "${SDK}/lib")
target_link_libraries(LeiaProbeMathCheck PRIVATE SimulatedRealityCore SimulatedRealityFaceTrackers psapi "${SDK}/third_party/OpenCV/lib/x64/opencv_world343.lib")
